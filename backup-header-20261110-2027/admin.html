Setup Reference & Scripts

Database SQL:
supabase-setup.sql 
-- 1. Create tables
CREATE TABLE IF NOT EXISTS public.adverts (
    id TEXT PRIMARY KEY,
    tag TEXT NOT NULL DEFAULT 'What''s new',
    title TEXT NOT NULL,
    body TEXT NOT NULL,
    image_url TEXT,
    button_label TEXT,
    button_url TEXT,
    published INTEGER NOT NULL DEFAULT 1,
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now()),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now())
);

CREATE TABLE IF NOT EXISTS public.accessories (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    category TEXT NOT NULL,
    description TEXT NOT NULL DEFAULT '',
    image_url TEXT NOT NULL,
    price NUMERIC(10, 2),
    currency TEXT NOT NULL DEFAULT 'USD',
    is_active INTEGER NOT NULL DEFAULT 1,
    sort_order INTEGER NOT NULL DEFAULT 0,
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now()),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now())
);

-- 2. Enable Row Level Security (RLS)
ALTER TABLE public.adverts ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.accessories ENABLE ROW LEVEL SECURITY;

-- 3. Public Read Policies
CREATE POLICY "Allow public read adverts" ON public.adverts FOR SELECT USING (true);
CREATE POLICY "Allow public read accessories" ON public.accessories FOR SELECT USING (true);

-- 4. Full Access for Service Role / Admin
CREATE POLICY "Full access adverts" ON public.adverts FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Full access accessories" ON public.accessories FOR ALL USING (true) WITH CHECK (true);

-- 5. Storage Bucket: learnotech-media
INSERT INTO storage.buckets (id, name, public, file_size_limit, allowed_mime_types)
VALUES ('learnotech-media', 'learnotech-media', true, 5242880, ARRAY['image/jpeg', 'image/png', 'image/webp'])
ON CONFLICT (id) DO UPDATE SET public = true;

CREATE POLICY "Public Access learnotech-media" ON storage.objects FOR SELECT USING (bucket_id = 'learnotech-media');
CREATE POLICY "Write Access learnotech-media" ON storage.objects FOR ALL USING (bucket_id = 'learnotech-media') WITH CHECK (bucket_id = 'learnotech-media');
